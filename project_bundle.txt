# --- PERSISTENT MEMORY INJECTION ---
BASE_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
ENV_PATH = os.path.join(BASE_DIR, ".env")


def load_env_file():
    """Load environment variables from a local .env file if present."""
    if not os.path.exists(ENV_PATH):
        return

    with open(ENV_PATH, "r", encoding="utf-8") as env_file:
        for line in env_file:
            stripped = line.strip()
            if not stripped or stripped.startswith("#") or "=" not in stripped:
                continue
            key, value = stripped.split("=", 1)
            os.environ.setdefault(key.strip(), value.strip().strip('"').strip("'"))


def get_env_value(name: str, default: str = "") -> str:
    """Read a value from environment or a local .env file."""
    load_env_file()
    return os.getenv(name, default)


def load_memory_content():
    """Load persistent assistant memory from memory.md for startup context."""
    memory_path = os.path.join(BASE_DIR, "memory.md")
    if os.path.exists(memory_path):
        try:
            with open(memory_path, "r", encoding="utf-8") as f:
                content = f.read().strip()
            return content if content else "No persistent memory found."
        except Exception:
            return "No persistent memory found."
    return "No persistent memory found."


initial_memory = load_memory_content()

def extract_user_preferences_from_message(message: str) -> dict:
    """Extract important user preferences from natural-language requests."""
    text = (message or "").strip()
    if not text:
        return {}

    preferences = {}
    lower = text.lower()

    if re.search(r"\b(prefer|like|love|want|need|enjoy)\b", lower):
        matches = re.findall(r"(?:prefer|like|love|want|need|enjoy)\s+(?:to\s+)?([^.;!?]+)", text, re.IGNORECASE)
        if matches:
            preferences["preference_summary"] = " ".join(part.strip() for part in matches if part.strip())

    colors = []
    if re.search(r"\bblue\b|\bgreen\b|\bred\b|\bblack\b|\bwhite\b|\bgold\b|\bpurple\b|\borange\b|\bpink\b", lower):
        colors = re.findall(r"\b(?:blue|green|red|black|white|gold|purple|orange|pink)\b", text, re.IGNORECASE)
        if colors:
            preferences["colors"] = ", ".join(dict.fromkeys(color.lower() for color in colors))

    styles = []
    if re.search(r"\b(premium|luxury|minimal|calm|bold|modern|playful|elegant|clean)\b", lower):
        styles = re.findall(r"\b(?:premium|luxury|minimal|calm|bold|modern|playful|elegant|clean)\b", text, re.IGNORECASE)

    if styles or colors:
        combined = [style.lower() for style in styles] + [color.lower() for color in colors]
        if combined:
            preferences["visual_style"] = ", ".join(dict.fromkeys(combined))

    if re.search(r"\b(branding|logo|caption|landing page|social media|website|design|product)\b", lower):
        preferences["interest_area"] = "creative design"

    return preferences

def learn_from_analytics(analytics: dict | str, campaign_name: str | None = None) -> str:
    """Persist analytics results and convert them into reusable content strategy memory for future prompts."""
    payload = _normalize_analytics_payload(analytics)
    if not payload:
        return "No analytics data was provided, so no learning memory was stored."

    brand_name = str(payload.get("brand_name") or campaign_name or payload.get("campaign") or "brand").strip() or "brand"
    platform = str(payload.get("platform") or "instagram").strip() or "instagram"
    summary = str(payload.get("summary") or "Performance summary not provided.").strip()
    reach = payload.get("reach", 0)
    likes = payload.get("likes", 0)
    comments = payload.get("comments", 0)
    saves = payload.get("saves", 0)
    shares = payload.get("shares", 0)
    engagement_rate = payload.get("engagement_rate")
    engagement_score = payload.get("engagement_score")

    if engagement_rate is None:
        try:
            engagement_rate = round(((int(likes) + int(comments) + int(saves) + int(shares)) / max(int(reach), 1) * 100), 2)
        except Exception:
            engagement_rate = 0.0
    if engagement_score is None:
        try:
            engagement_score = round((int(likes) + int(comments) + int(saves) + int(shares)) / 200, 2)
        except Exception:
            engagement_score = 0.0

    learning_record = {
        "brand_name": brand_name,
        "platform": platform,
        "reach": int(reach),
        "likes": int(likes),
        "comments": int(comments),
        "saves": int(saves),
        "shares": int(shares),
        "engagement_rate": float(engagement_rate),
        "engagement_score": float(engagement_score),
        "summary": summary,
        "recorded_at": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
    }

    key = f"analytics:{brand_name.lower().replace(' ', '_')}:{learning_record['recorded_at'].replace(' ', '_')}"
    remember("analytics", key, json.dumps(learning_record, ensure_ascii=False))

    profile = {}
    profile_raw = get_memory_entry("user_profile", "")
    if profile_raw:
        try:
            profile = json.loads(profile_raw)
        except Exception:
            profile = {}

    profile["last_engagement_summary"] = (
        f"{brand_name} on {platform}: engagement score {engagement_score}, engagement rate {engagement_rate}%."
    )
    if float(engagement_rate) >= 8.0:
        profile["content_strategy_hint"] = (
            f"Strong performer: keep the premium, bold visual style and clear CTA for {brand_name} on {platform}."
        )
    else:
        profile["content_strategy_hint"] = (
            f"Needs refinement: test a clearer hook, stronger CTA, and more consistent brand colors for {brand_name} on {platform}."
        )

    remember("profile", "user_profile", json.dumps(profile, ensure_ascii=False))
    track_topic_context(brand_name, summary or f"Engagement score {engagement_score} on {platform}.")

    response = {
        "brand_name": brand_name,
        "platform": platform,
        "engagement_rate": float(engagement_rate),
        "engagement_score": float(engagement_score),
        "summary": summary,
        "memory_saved": True,
        "status": "learning-memory-updated",
    }
    return json.dumps(response, ensure_ascii=False, indent=2)

def _build_conversation_summary(history: list[dict]) -> str:
    """Create a short compatibility summary of recent user facts so communication remains coherent over long chats."""
    user_facts = []
    for message in history:
        if not isinstance(message, dict):
            continue
        if message.get("role") != "user":
            continue
        content = _clean_message_text(message)
        if not content:
            continue
        user_facts.append(content)

    saved_summary = get_memory_entry("conversation_summary", "")
    if user_facts:
        recent_facts = user_facts[-4:]
        summary = "User context: " + "; ".join(recent_facts)
        remember("profile", "conversation_summary", summary)
        return summary

    return saved_summary or "No prior user preferences captured yet."


def trim_and_summarize_conversation(history: list[dict]) -> list[dict]:
    """Trim long conversation logs and keep a concise summary plus recent turns to preserve continuity."""
    if not history:
        return [{"role": "system", "content": "No conversation context yet."}]

    summary = _build_conversation_summary(history)
    recent_messages = []
    for message in history[-8:]:
        if isinstance(message, dict):
            content = _clean_message_text(message)
            if content:
                recent_messages.append(message)

    final_history = [{"role": "system", "content": summary}]
    final_history.extend(recent_messages)
    return final_history[:12]

def call_antigravity_cli(messages: list) -> str:
    """Route the recent conversation through the local AGY CLI using a compact prompt format."""
    if not AGY_PATH or not os.path.exists(AGY_PATH):
        raise RuntimeError(f"Antigravity CLI not found at {AGY_PATH or 'PATH/LOCALAPPDATA'}")

    system_prompt = ""
    dialogue = []
    for msg in messages:
        role = msg.get("role", "")
        content = msg.get("content", "")
        if not content:
            continue
        if role == "system":
            system_prompt = content
        elif role == "user":
            dialogue.append(f"User: {content}")
        elif role == "assistant":
            dialogue.append(f"Assistant: {content}")

    recent_dialogue = "\n".join(dialogue[-10:])
    full_prompt = (
        f"System Context:\n{system_prompt}\n\n"
        f"Conversation:\n{recent_dialogue}\n\n"
        "Respond concisely as JARVIS to the latest User message."
    )

    command_variants = [
        [AGY_PATH, "--dangerously-skip-permissions", "--add-dir", BASE_DIR, "-p", full_prompt],
        [AGY_PATH, "--dangerously-skip-permissions", "-p", full_prompt],
        [AGY_PATH, "-p", full_prompt],
        [AGY_PATH, "--prompt", full_prompt],
    ]

    last_error = ""
    for command in command_variants:
        try:
            result = subprocess.run(
                command,
                cwd=BASE_DIR,
                capture_output=True,
                text=True,
                encoding="utf-8",
                timeout=90,
            )
            output = (result.stdout or result.stderr or "").strip()
            if result.returncode == 0 and output:
                return output
            last_error = output or f"agy exited with code {result.returncode}"
            lower_output = output.lower()
            if "not recognized" in lower_output or "unknown option" in lower_output:
                continue
        except (FileNotFoundError, OSError, subprocess.TimeoutExpired) as exc:
            last_error = str(exc)
            continue

    raise RuntimeError(last_error or "Antigravity CLI failed without output.")


def call_cloudflare_workers_ai(messages: list) -> str:
    """Routes chat completions through Cloudflare Workers AI using the OpenAI compatibility layer."""
    if OpenAI is None:
        raise RuntimeError("openai package is not installed; Cloudflare gateway cannot be used.")

    account_id = os.getenv("CLOUDFLARE_ACCOUNT_ID")
    api_token = os.getenv("CLOUDFLARE_API_TOKEN")
    if not account_id or not api_token:
        raise RuntimeError("Cloudflare credentials missing in .env")

    client = OpenAI(
        base_url=f"https://api.cloudflare.com/client/v4/accounts/{account_id}/ai/v1",
        api_key=api_token,
    )

    response = client.chat.completions.create(
        model="@cf/meta/llama-3.3-70b-instruct-fp8-fast",

        timeout=30,
    )

    content = response.choices[0].message.content
    return (content or "").strip()

def _normalize_host(url: str) -> str:
    """Return the host for a URL, stripped of protocol and www prefix."""
    if not url:
        return ""
    candidate = url if "://" in url else f"https://{url}"
    parsed = urlparse(candidate)
    return parsed.netloc.lower().replace("www.", "")


def _is_approved_domain(url: str, allowlist: list[str] | None = None) -> bool:
    """Check whether the target URL is inside the approved browser allowlist."""
    allow = {item.lower().replace("www.", "") for item in (allowlist or DEFAULT_BROWSER_ALLOWLIST)}
    host = _normalize_host(url)
    if not host:
        return False
    return host in allow or any(host.endswith(f".{domain}") for domain in allow)

def capture_engagement_summary(
    reach: int = 0,
    likes: int = 0,
    comments: int = 0,
    saves: int = 0,
    shares: int = 0,
) -> str:
    """Summarize post performance with a compact engagement score for review and optimization."""
    reach_total = max(int(reach), 0)
    likes_total = max(int(likes), 0)
    comments_total = max(int(comments), 0)
    saves_total = max(int(saves), 0)
    shares_total = max(int(shares), 0)
    total_interactions = likes_total + comments_total + saves_total + shares_total
    engagement_rate = round((total_interactions / reach_total * 100), 2) if reach_total else 0.0
    engagement_index = round(total_interactions / 200, 2)

    result = {
        "reach": reach_total,
        "likes": likes_total,
        "comments": comments_total,
        "saves": saves_total,
        "shares": shares_total,
        "engagement_rate": engagement_rate,
        "engagement_score": engagement_index,
        "summary": (
            f"Engagement score: {engagement_index:.1f}. "
            f"{total_interactions} interactions from {reach_total} reach "
            f"produced an engagement rate of {engagement_rate:.2f}%."
        ),
        "status": "insights-ready",
    }
    return json.dumps(result, ensure_ascii=False, indent=2)

    )

def log_long_term_memory(category: str, content: str) -> str:
    """Save a structured note, preference, or project milestone to the database."""
    try:
        init_db()
        conn = sqlite3.connect(DB_PATH)
        cursor = conn.cursor()
        timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        cursor.execute(
            "INSERT INTO long_term_memory (timestamp, category, content) VALUES (?, ?, ?)",
            (timestamp, category, content),
        )
        conn.commit()
        conn.close()
        return f"Successfully logged memory under category '{category}'."
    except Exception as e:
        return f"Error logging memory: {e}"


def search_long_term_memory(query: str) -> str:

    try:
        init_db()
        conn = sqlite3.connect(DB_PATH)
        cursor = conn.cursor()
        cursor.execute(
            "SELECT timestamp, category, content FROM long_term_memory WHERE content LIKE ? OR category LIKE ? ORDER BY id DESC",

        )

        conn.close()

        results = [f"[{row[0]}] ({row[1]}): {row[2]}" for row in rows]
        return "Found memories:\n" + "\n".join(results)
    except Exception as e:
        return f"Error searching memory: {e}"

def listen_to_user(hud_callback=None) -> str:
    """Captures microphone input and converts it to text safely."""
    recognizer = sr.Recognizer()
    
    try:
        with sr.Microphone() as source:
            log_info("Calibrating microphone for ambient noise...")
            if hud_callback:
                hud_callback("CALIBRATING MIC...", "listening")
            
            recognizer.adjust_for_ambient_noise(source, duration=0.8)
            
            log_info("Listening for command...")
            if hud_callback:
                hud_callback("LISTENING // SPEAK NOW", "listening")
            
            # Listen with a timeout so it doesn't freeze permanently
            audio = recognizer.listen(source, timeout=6, phrase_time_limit=10)
            
            if hud_callback:
                hud_callback("PROCESSING SPEECH...", "thinking")
                
            query = recognizer.recognize_google(audio)
            log_info(f"Recognized speech: {query}")
            return query
            
    except sr.WaitTimeoutError:
        log_info("Listening timed out. No speech detected.")
        return ""
    except sr.UnknownValueError:
        log_info("Could not understand audio.")
        return ""
    except Exception as e:
        log_error(f"Microphone error: {e}")
        return ""

def _get_gateway_token() -> str:
    return str(os.getenv("JARVIS_API_TOKEN", "jarvis-local-dev")).strip()


def _get_rate_limit_config() -> tuple[int, int]:
    try:
        limit = max(1, int(os.getenv("JARVIS_RATE_LIMIT", "120")))
    except ValueError:
        limit = 120
    try:
        window_seconds = max(1, int(os.getenv("JARVIS_RATE_WINDOW_SECONDS", "60")))
    except ValueError:
        window_seconds = 60
    return limit, window_seconds


def _check_rate_limit(client_ip: str) -> bool:
    limit, window_seconds = _get_rate_limit_config()
    now = time.time()
    with _GATEWAY_REQUESTS_LOCK:
        bucket = _GATEWAY_REQUESTS.setdefault(client_ip, [])
        bucket[:] = [timestamp for timestamp in bucket if now - timestamp < window_seconds]
        if len(bucket) >= limit:
            return False
        bucket.append(now)
    return True


def _ensure_gateway_logger() -> logging.Logger:
    logger = logging.getLogger("jarvis.gateway")
    if logger.handlers:
        return logger
    os.makedirs(os.path.dirname(_GATEWAY_LOG_PATH), exist_ok=True)
    handler = logging.FileHandler(_GATEWAY_LOG_PATH, encoding="utf-8")
    handler.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(message)s"))
    logger.addHandler(handler)
    logger.setLevel(logging.INFO)
    logger.propagate = False
    return logger


def _gateway_auth_error() -> tuple[dict, int]:
    return {"error": "Unauthorized: valid Authorization: Bearer <token> is required."}, 401


def _gateway_rate_limit_error() -> tuple[dict, int]:
    return {"error": "Rate limit exceeded: too many requests from this client."}, 429


async def _apply_gateway_security(request, call_next):
    logger = _ensure_gateway_logger()
    client_ip = request.client.host if request.client else "unknown"
    token = _get_gateway_token()
    auth_header = request.headers.get("authorization", "")
    if token:
        parts = auth_header.split(None, 1)
        if len(parts) != 2 or parts[0].lower() != "bearer" or parts[1] != token:
            logger.warning("Rejected unauthorized request from %s for %s", client_ip, request.url.path)
            return JSONResponse(*_gateway_auth_error())

    if not _check_rate_limit(client_ip):
        logger.warning("Rate limit triggered for %s on %s", client_ip, request.url.path)
        return JSONResponse(*_gateway_rate_limit_error())

    logger.info("%s %s from %s", request.method, request.url.path, client_ip)
    return await call_next(request)

        self._register_route("/tools", "GET", lambda: {"tools": list_mcp_tools()})
        self._register_route("/memory", "GET", lambda: {"context": get_shared_memory_context()})

    def _register_route(self, path: str, method: str, handler):
        self.routes.append(type("Route", (), {"path": path, "method": method, "handler": handler})())

    def route(self, path: str, methods=None):
        def decorator(func):
            self._register_route(path, ",".join(methods or ["GET"]), func)
            return func
        return decorator

def handle_mcp_message(message: dict) -> dict:
    """Process JSON-RPC-like MCP requests for tools and workspace operations."""
    request_id = message.get("id", 1)
    method = message.get("method")
    params = message.get("params", {})

    if method in ("tools/list", "list"):
        return {"jsonrpc": "2.0", "id": request_id, "result": {"tools": list_mcp_tools()}}

    if method in ("tools/call", "call"):
        tool_name = params.get("name") or params.get("tool")
        arguments = params.get("arguments", params.get("args", {}))
        try:
            result = call_mcp_tool(tool_name, arguments)
            response = {
                "jsonrpc": "2.0",
                "id": request_id,
                "result": {
                    "content": [{"type": "text", "text": result}],
                    "name": tool_name,
                },
            }
        except Exception as exc:
            response = {"jsonrpc": "2.0", "id": request_id, "error": {"message": str(exc), "code": -32601}}
        return response

    if method == "workspace/read":
        path = params.get("path") or params.get("filepath")
        try:
            text = read_text_file(path)
            return {"jsonrpc": "2.0", "id": request_id, "result": {"content": text}}
        except Exception as exc:
            return {"jsonrpc": "2.0", "id": request_id, "error": {"message": str(exc), "code": -32000}}

    if method == "workspace/write":
        path = params.get("path") or params.get("filepath")
        content = params.get("content", "")
        try:
            result = write_local_file(path, content, confirmed=params.get("confirmed", True))
            return {"jsonrpc": "2.0", "id": request_id, "result": {"content": result}}
        except Exception as exc:
            return {"jsonrpc": "2.0", "id": request_id, "error": {"message": str(exc), "code": -32000}}

    if method == "workspace/list":

class MCPRequestHandler(BaseHTTPRequestHandler):
    """Minimal HTTP JSON-RPC-style MCP server for local AI clients."""

    def do_GET(self):
        if self.path == "/health":
            payload = {"status": "ok", "server": "jarvis-mcp"}
            self._send_json(payload)
            return
        self._send_json({"error": "not found"}, status=404)

    def do_POST(self):
        length = int(self.headers.get("Content-Length", "0"))
        raw_body = self.rfile.read(length)
        try:
            payload = json.loads(raw_body.decode("utf-8")) if raw_body else {}
        except json.JSONDecodeError:
            self._send_json({"error": "invalid JSON body"}, status=400)
            return

        response = handle_mcp_message(payload)
        self._send_json(response)

    def _send_json(self, payload: dict, status: int = 200):
        body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, format: str, *args: Any):
        return

import os
import tempfile
import threading
from contextlib import contextmanager

from config.settings import BASE_DIR

_LOCKS = {}
_LOCKS_GUARD = threading.Lock()


def project_safe_path(filepath: str) -> str:
    """Resolve a project-local path and reject attempts to escape the workspace root.

    All model-driven reads and writes are intentionally permitted anywhere under the
    JARVIS project directory, but never outside it.
    """
    if not filepath or not str(filepath).strip():
        raise ValueError("A file path is required.")

    workspace_root = os.path.realpath(BASE_DIR)
    candidate = os.path.realpath(
        os.path.normpath(filepath if os.path.isabs(filepath) else os.path.join(BASE_DIR, filepath))
    )

    if os.path.commonpath([workspace_root, candidate]) != workspace_root:
        raise ValueError("Access denied: path escapes the project directory.")

    return candidate


@contextmanager
def file_lock(path: str):
    """Cross-thread lock guard for atomic transactional reads and writes."""
    safe_path = project_safe_path(path)
    with _LOCKS_GUARD:
        lock = _LOCKS.setdefault(safe_path, threading.RLock())
    with lock:
        yield


def atomic_write_text(filepath: str, content: str, encoding: str = "utf-8") -> str:
    """Atomically writes text to a project-scoped file using a temp file + replace pattern."""
    safe_path = project_safe_path(filepath)
    directory = os.path.dirname(safe_path) or BASE_DIR
    os.makedirs(directory, exist_ok=True)

    with file_lock(filepath):
        fd, temp_path = tempfile.mkstemp(prefix=".jarvis-write-", suffix=".tmp", dir=directory)
        try:
            with os.fdopen(fd, "w", encoding=encoding) as handle:
                handle.write(content)
            os.replace(temp_path, safe_path)
        except Exception:
            if os.path.exists(temp_path):
                os.remove(temp_path)
            raise

    return safe_path


def read_text_file(filepath: str, encoding: str = "utf-8") -> str:
    """Safely read text from a project-scoped file under a file lock."""
    safe_path = project_safe_path(filepath)
    with file_lock(filepath):
        with open(safe_path, "r", encoding=encoding, errors="ignore") as handle:
            return handle.read()


def list_project_directory(directory_path: str = ".") -> list[str]:
    """List the visible contents of a project directory in a stable, safe manner."""
    safe_path = project_safe_path(directory_path)
    if not os.path.isdir(safe_path):
        raise FileNotFoundError(f"Directory does not exist: {directory_path}")
    return sorted(os.listdir(safe_path))


def write_and_backup(filepath: str, content: str, encoding: str = "utf-8") -> dict:
    """Write content with a backup snapshot, returning metadata for tooling and auditing."""
    safe_path = project_safe_path(filepath)
    backup_path = None
    if os.path.exists(safe_path):
        backup_path = safe_path + ".bak"
        with open(safe_path, "r", encoding=encoding, errors="ignore") as src:
            payload = src.read()
        with open(backup_path, "w", encoding=encoding) as dst:
            dst.write(payload)

    atomic_write_text(filepath, content, encoding=encoding)
    return {
        "path": safe_path,
        "backup": backup_path,
        "bytes_written": len(content.encode(encoding)),
    }

class JarvisScheduler:

    self.briefing_hour = briefing_hour
    self.briefing_minute = briefing_minute
    self.hud_callback = hud_callback
    self._stop_event = threading.Event()
    self.thread = threading.Thread(
        target=self._run_scheduler, daemon=True, name="JarvisSchedulerThread"
    )

  def start(self):
    """Starts the background scheduler loop."""
    log_info("Starting JARVIS background event scheduler...")
    self.thread.start()

  def stop(self):
    """Stops the background scheduler thread."""
    self._stop_event.set()

    """Gathers data, queries local LLM, and speaks the morning briefing."""

        f" systems are operating at peak efficiency, and outline a productive"

    )

    return backup_path


def rollback_file(filepath: str, backup_path: str = "") -> str:
    """Restore a file from a specified backup or the most recent automatic backup."""
    safe_path = _resolve_project_path(filepath)
    rel_path = os.path.relpath(safe_path, PROJECT_ROOT)
    sanitized_name = rel_path.replace(os.sep, "__").replace("/", "__")

    target_backup = backup_path
    if not target_backup:
        pointer_file = os.path.join(BACKUP_DIR, f"latest__{sanitized_name}.txt")
        if os.path.exists(pointer_file):
            with open(pointer_file, "r", encoding="utf-8") as f:
                target_backup = f.read().strip()

    if not target_backup or not os.path.exists(target_backup):
        # Fallback search for latest file matching sanitized pattern
        if os.path.exists(BACKUP_DIR):
            candidates = sorted(
                [
                    os.path.join(BACKUP_DIR, f)
                    for f in os.listdir(BACKUP_DIR)
                    if f.endswith(f"__{sanitized_name}")
                ],
                reverse=True,
            )
            if candidates:
                target_backup = candidates[0]

    if not target_backup or not os.path.exists(target_backup):
        return f"Error: No backup found to restore for {filepath}."

    os.makedirs(os.path.dirname(safe_path), exist_ok=True)
    shutil.copy2(target_backup, safe_path)
    log_info(f"Restored {safe_path} from backup {target_backup}")
    return f"Successfully rolled back {filepath} from {os.path.basename(target_backup)}."

        if not valid:
            return f"Self-improvement aborted before writing: {syntax_msg}"

    # Step 2: Backup existing file
    backup_path = ""
    if os.path.exists(safe_path):
        backup_path = create_backup(filepath)

    # Step 3: Apply the file modification
    try:
        os.makedirs(os.path.dirname(safe_path), exist_ok=True)
        with open(safe_path, "w", encoding="utf-8") as f:
            f.write(new_content)
    except Exception as write_err:
        return f"Failed to write file {filepath}: {write_err}"

    # Step 4: Verification via test suite
    if run_tests and is_python:
        # Check file compilation on disk
        valid_disk, disk_err = validate_code_syntax(filepath=filepath)
        if not valid_disk:
            rollback_status = rollback_file(filepath, backup_path) if backup_path else "No backup available"
            return f"Self-improvement aborted and rolled back due to disk compilation error: {disk_err}. {rollback_status}"

        passed, test_output = run_project_tests()
        if not passed:
            rollback_status = rollback_file(filepath, backup_path) if backup_path else "No backup available"

                f"Self-improvement failed verification tests and was automatically rolled back!\n"
                f"Test failure output:\n{test_output[:600]}\n"
                f"Rollback status: {rollback_status}"
            )

    # Step 5: Post-success dynamic reloading if module was already loaded
    rel_path = os.path.relpath(safe_path, PROJECT_ROOT)
    if rel_path.endswith(".py"):
        mod_candidate = rel_path[:-3].replace(os.sep, ".").replace("/", ".")
        if mod_candidate in sys.modules:
            reload_jarvis_module(mod_candidate)

    log_info(f"Self-improvement successfully committed to {filepath}")
    return f"Successfully applied self-improvement to {filepath}. Tests verified and passed."

def _resolve_project_path(filepath: str) -> str:
  """Return a project-scoped absolute path while allowing full access inside the JARVIS tree."""
  if not filepath or not filepath.strip():
    raise ValueError("A file path is required.")

    raise ValueError("Access denied: path escapes the project directory.")

def web_search(query: str, max_results: int = 4) -> str:
  """Searches the live web via Tavily and returns concise, cited results."""
  api_key = os.getenv("TAVILY_API_KEY")
  if not api_key:
    return "Error: TAVILY_API_KEY missing in .env"

  payload = json.dumps({
      "query": query,
      "search_depth": "basic",
      "max_results": int(max_results) if max_results and max_results > 0 else 4,
  }).encode("utf-8")

  req = urllib.request.Request(
      "https://api.tavily.com/search",
      data=payload,
      headers={
          "Authorization": f"Bearer {api_key}",
          "Content-Type": "application/json",
      },
      method="POST",
  )

  try:
      with urllib.request.urlopen(req, timeout=20) as resp:
          data = json.loads(resp.read().decode("utf-8"))
      results = data.get("results", [])
      if not results:
          return f"No web results found for '{query}'."
      return "\n".join(
          f"- {r.get('title', 'Untitled')} ({r.get('url', '')}): {r.get('content', '')}"
          for r in results[: max_results if max_results and max_results > 0 else 4]
      )
  except Exception as e:
      return f"Web search error: {e}"

def search_project_files(query: str, directory_path: str = ".", max_results: int = 20) -> str:
  """Search for filenames or content matches inside the project tree."""
  try:
    safe_root = _resolve_project_path(directory_path)
    matches = []
    query_text = query.lower()

    for root, dirnames, filenames in os.walk(safe_root):
      dirnames[:] = [d for d in dirnames if not d.startswith(".") and d != "__pycache__"]

      for name in sorted(dirnames + filenames):
        full_path = os.path.join(root, name)
        rel_path = os.path.relpath(full_path, PROJECT_ROOT)
        if query_text in name.lower():
          matches.append(rel_path)
          if len(matches) >= max_results:
            break

      if len(matches) >= max_results:
        break

      for filename in sorted(filenames):
        full_path = os.path.join(root, filename)
        if len(matches) >= max_results:
          break
        try:
          with open(full_path, "r", encoding="utf-8", errors="ignore") as handle:
            content = handle.read().lower()
          if query_text in content:
            rel_path = os.path.relpath(full_path, PROJECT_ROOT)
            matches.append(rel_path)
        except Exception:
          continue

    if not matches:
      return f"No files or content matching '{query}' were found under {directory_path}."

    return "Matches:\n" + "\n".join(matches[:max_results])
  except Exception as e:
    return f"Error searching project files: {e}"

import hashlib
import json
import os
import re
import sqlite3
from collections import Counter
from datetime import datetime

from config.settings import BASE_DIR

VECTOR_DB_PATH = os.path.join(BASE_DIR, "memory", "jarvis_vector_memory.db")
_VECTOR_DIMENSIONS = 128


def _ensure_vector_db() -> None:
    os.makedirs(os.path.dirname(VECTOR_DB_PATH), exist_ok=True)
    conn = sqlite3.connect(VECTOR_DB_PATH)
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS vector_memory (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            category TEXT,
            content TEXT,
            embedding TEXT,
            created_at TEXT
        )
        """
    )
    conn.commit()
    conn.close()


def _tokenize(text: str) -> list[str]:
    if not text:
        return []
    tokens = re.findall(r"[a-zA-Z0-9_]+", str(text).lower())
    return [token for token in tokens if len(token) > 1]


def _build_embedding(text: str, dimensions: int = _VECTOR_DIMENSIONS) -> list[float]:
    counts = Counter(_tokenize(text))
    vector = [0.0] * dimensions
    if not counts:
        return vector

    for token, weight in counts.items():
        index = int(hashlib.sha256(token.encode("utf-8")).hexdigest(), 16) % dimensions
        vector[index] += float(weight)

    magnitude = (sum(value * value for value in vector)) ** 0.5
    if magnitude:
        vector = [value / magnitude for value in vector]
    return vector


def _cosine_similarity(left: list[float], right: list[float]) -> float:
    if not left or not right or len(left) != len(right):
        return 0.0

    dot = sum(a * b for a, b in zip(left, right))
    left_norm = (sum(value * value for value in left)) ** 0.5
    right_norm = (sum(value * value for value in right)) ** 0.5
    if left_norm == 0 or right_norm == 0:
        return 0.0
    return dot / (left_norm * right_norm)


def save_vector_memory(content: str, category: str = "general") -> dict:
    """Persist a memory fragment as a vector so it can be semantically matched later."""
    if not content or not str(content).strip():
        raise ValueError("content is required")

    _ensure_vector_db()
    embedding = _build_embedding(content)
    created_at = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    conn = sqlite3.connect(VECTOR_DB_PATH)
    cursor = conn.cursor()
    cursor.execute(
        "INSERT INTO vector_memory (category, content, embedding, created_at) VALUES (?, ?, ?, ?)",
        (category, content.strip(), json.dumps(embedding), created_at),
    )
    conn.commit()
    record_id = cursor.lastrowid
    conn.close()
    return {"id": record_id, "category": category, "content": content.strip(), "score": 1.0}


def semantic_search(query: str, limit: int = 5, category: str | None = None) -> list[dict]:
    """Find semantically related memory entries using cosine similarity over hashed vectors."""
    if not query or not str(query).strip():
        return []

    _ensure_vector_db()
    query_embedding = _build_embedding(query)

    conn = sqlite3.connect(VECTOR_DB_PATH)
    if category:
        rows = conn.execute(
            "SELECT id, category, content, embedding FROM vector_memory WHERE category = ? ORDER BY id DESC",
            (category,),
        ).fetchall()
    else:
        rows = conn.execute(
            "SELECT id, category, content, embedding FROM vector_memory ORDER BY id DESC"
        ).fetchall()
    conn.close()

    matches = []
    for record_id, item_category, item_content, embedding_blob in rows:
        try:
            stored_vector = json.loads(embedding_blob)
        except (TypeError, ValueError):
            continue
        score = _cosine_similarity(query_embedding, stored_vector)
        if score > 0.0:
            matches.append(
                {
                    "id": record_id,
                    "category": item_category,
                    "content": item_content,
                    "score": round(float(score), 4),
                }
            )

    matches.sort(key=lambda item: item["score"], reverse=True)
    return matches[: max(1, int(limit))]

    print(f"\n[VOICE MUTED] JARVIS: {text}")
    if hud_callback:
      hud_callback("VOICE MUTED", "idle", text)
    return

  print(f"\nJARVIS: {text}")
  if hud_callback:
    hud_callback("SPEAKING", "speaking", text)

  with speech_lock:
    try:
      if kokoro_pipeline and sd is not None:
        generator = kokoro_pipeline(text, voice=VOICE_ID, speed=1.0)

          sd.wait()

        fallback_engine.say(text)
        fallback_engine.runAndWait()
      else:
        print(f"[VOICE UNAVAILABLE] JARVIS: {text}")
    except Exception as e:
      log_error(f"Kokoro audio error: {e}. Falling back to system TTS.")
      if fallback_engine is not None:
        fallback_engine.say(text)

def _slugify(value: str) -> str:
    cleaned = re.sub(r"[^a-zA-Z0-9._-]+", "-", (value or "experiment").strip())
    cleaned = cleaned.strip(".-")
    return cleaned or "experiment"


def ensure_experiment_directory(name: str, base_dir: str = BASE_DIR) -> str:
    """Create a project-scoped experiment folder and a starter README for model experiments."""
    slug = _slugify(name)
    stamp = datetime.now().strftime("%Y%m%d-%H%M%S")
    target_dir = os.path.join(base_dir, "experiments", f"{slug}_{stamp}")
    os.makedirs(target_dir, exist_ok=True)

    readme_path = os.path.join(target_dir, "README.md")
    if not os.path.exists(readme_path):
        with open(readme_path, "w", encoding="utf-8") as handle:
            handle.write(
                f"# {slug}\n\n"
                f"Created: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n\n"
                "This folder is reserved for project experiments, prototypes, and model-driven testing.\n"
            )

    return target_dir


def append_workspace_growth_log(decision: str, context: str = "") -> str:
    """Append a model decision to the shared project log for traceable growth and review."""
    if not decision or not str(decision).strip():
        raise ValueError("decision text is required")

    os.makedirs(os.path.dirname(WORKSPACE_GROWTH_LOG), exist_ok=True)
    entry = {
        "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        "context": context or "general",
        "decision": str(decision).strip(),
    }

    with open(WORKSPACE_GROWTH_LOG, "a", encoding="utf-8") as handle:
        handle.write(json.dumps(entry, ensure_ascii=False) + "\n")

    os.makedirs(os.path.dirname(WORKSPACE_GROWTH_MD), exist_ok=True)
    with open(WORKSPACE_GROWTH_MD, "a", encoding="utf-8") as handle:
        handle.write(f"\n## {entry['timestamp']}\n**Context:** {entry['context']}\n**Decision:** {entry['decision']}\n")

    return f"Workspace growth decision logged: {entry['timestamp']}"


def search_workspace_growth_log(query: str, limit: int = 10) -> list[dict]:
    """Search the shared workspace decision log for relevant model decisions."""
    if not os.path.exists(WORKSPACE_GROWTH_LOG):
        return []
    if limit <= 0:
        limit = 10

    needle = (query or "").lower()
    matches = []
    with open(WORKSPACE_GROWTH_LOG, "r", encoding="utf-8", errors="ignore") as handle:
        for line in handle:
            try:
                payload = json.loads(line)
            except json.JSONDecodeError:
                continue
            haystack = json.dumps(payload, ensure_ascii=False).lower()
            if not needle or needle in haystack:
                matches.append(payload)

    matches = matches[-limit:]
    return matches


def _ensure_project_memory_index_db() -> sqlite3.Connection:
    os.makedirs(os.path.dirname(PROJECT_MEMORY_INDEX_PATH), exist_ok=True)
    conn = sqlite3.connect(PROJECT_MEMORY_INDEX_PATH)
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS project_memory_index (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            source TEXT,
            category TEXT,
            content TEXT,
            created_at TEXT,
            keywords TEXT,
            digest TEXT UNIQUE
        )
        """
    )
    return conn


def _keyword_tokens(value: str) -> list[str]:
    return [token for token in re.findall(r"[A-Za-z0-9_]+", (value or "").lower()) if len(token) > 2]


def build_project_memory_index(force: bool = False) -> list[dict]:
    """Build a project-wide searchable memory index from the primary memory DB and vector memory records."""
    conn = _ensure_project_memory_index_db()
    if force:
        conn.execute("DELETE FROM project_memory_index")

    seen = set()
    rows = []

    primary_db = os.path.join(BASE_DIR, "memory", "jarvis_memory.db")
    if os.path.exists(primary_db):
        primary_conn = sqlite3.connect(primary_db)
        primary_rows = primary_conn.execute(
            "SELECT category, value, created_at FROM memory ORDER BY id DESC"
        ).fetchall()
        primary_conn.close()
        for category, value, created_at in primary_rows:
            if not value or not str(value).strip():
                continue
            digest = f"sqlite::{category}::{str(value).strip()}"
            if digest in seen:
                continue
            seen.add(digest)
            row = {
                "source": "sqlite_memory",
                "category": category or "general",
                "content": str(value).strip(),
                "created_at": created_at or datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                "keywords": json.dumps(_keyword_tokens(str(value)), ensure_ascii=False),
                "digest": digest,
            }
            rows.append(row)

    vector_db = os.path.join(BASE_DIR, "memory", "jarvis_vector_memory.db")
    if os.path.exists(vector_db):
        vector_conn = sqlite3.connect(vector_db)
        vector_rows = vector_conn.execute(
            "SELECT category, content, created_at FROM vector_memory ORDER BY id DESC"
        ).fetchall()
        vector_conn.close()
        for category, value, created_at in vector_rows:
            if not value or not str(value).strip():
                continue
            digest = f"vector::{category}::{str(value).strip()}"
            if digest in seen:
                continue
            seen.add(digest)
            row = {
                "source": "vector_memory",
                "category": category or "general",
                "content": str(value).strip(),
                "created_at": created_at or datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                "keywords": json.dumps(_keyword_tokens(str(value)), ensure_ascii=False),
                "digest": digest,
            }
            rows.append(row)

    for row in rows:
        conn.execute(
            "INSERT OR REPLACE INTO project_memory_index (source, category, content, created_at, keywords, digest) VALUES (?, ?, ?, ?, ?, ?)",
            (row["source"], row["category"], row["content"], row["created_at"], row["keywords"], row["digest"]),
        )

    conn.commit()
    conn.close()
    return rows


def search_project_memory_index(query: str, limit: int = 10, category: str | None = None) -> list[dict]:
    """Return best matches from the project-wide memory index for a query or category."""
    rows = build_project_memory_index(force=False)
    if not rows:
        return []

    query_tokens = {token for token in _keyword_tokens(query or "")}
    def score(item: dict) -> int:
        text = (item["content"] + " " + (item["category"] or "")).lower()
        if not query_tokens:
            return 1
        score = 0
        for token in query_tokens:
            if token in text:
                score += 2
        if query and query.lower() in text:
            score += 4
        if category and item["category"] == category:
            score += 3
        return score

    filtered = [item for item in rows if not category or item["category"] == category]
    filtered.sort(key=score, reverse=True)
    return [{
        "source": item["source"],
        "category": item["category"],
        "content": item["content"],
        "created_at": item["created_at"],
        "score": score(item),
    } for item in filtered[: max(1, int(limit))]]

def process_command(query: str, hud_callback=None):
  """Autonomous agent router leveraging LLM tool-calling and memory integration."""
  global hud_instance, scheduler
  q_lower = query.lower()

  if hud_callback:
    hud_callback("PROCESSING", "thinking", f"Autonomous analysis: '{query}'")

  # Check if voice feedback is enabled in the HUD settings panel
  speak_allowed = True
  if hud_instance and hasattr(hud_instance, "voice_enabled"):
    speak_allowed = hud_instance.voice_enabled

  response_text = ""
  try:
    # 1. Specialized Memory Saving Trigger
    if "remember" in q_lower or "save memory" in q_lower:
      content = query.replace("remember", "").replace("save memory", "").strip()
      result_msg = save_memory(content, category="user_preference")
      response_text = f"I have committed that to long-term memory, Sir. [{result_msg}]"
      if speak_allowed:
        speak("I have committed that to long-term memory, Sir.", hud_callback)

    # 2. Specialized Memory Recall Trigger
    elif "recall" in q_lower or "what do you remember" in q_lower:
      memories = recall_memories()
      if isinstance(memories, list) and memories:
        response_text = "\n".join([f"[{ts}] {q} -> {r}" if isinstance(r, str) else str(item) for item in memories for ts, q, r in [item] if len(item)==3]) if memories and len(memories[0])==3 else str(memories)
      else:
        response_text = str(memories)
      if speak_allowed:
        speak("Here are your stored long-term memories, Sir.", hud_callback)

    # 3. Morning Briefing Trigger
    elif "morning briefing" in q_lower:
      if scheduler:
        threading.Thread(
            target=scheduler.trigger_morning_briefing, daemon=True
        ).start()
        response_text = "Morning briefing initiated."
      else:
        response_text = "Scheduler is not active, Sir."
      if speak_allowed:
        speak(response_text, hud_callback)

    # 4. Autonomous Agent Loop (LLM Tool Calling)
    else:
      response_text = ask_local_brain_with_tools(query)
      if speak_allowed:
        speak(response_text, hud_callback)

    if hud_callback:
      hud_callback("STANDBY // READY", "idle", response_text)

    # Automatically save interaction to database history
    save_interaction(query, response_text)

  except Exception as e:
    err_msg = f"Autonomous error: {e}"
    log_error(err_msg)
    if speak_allowed:
      speak("I encountered an error processing that request, Sir.", hud_callback)
    if hud_callback:
      hud_callback("SYSTEM ERROR", "idle", err_msg)
    save_interaction(query, err_msg)

# J.A.R.V.I.S. Project Guide

J.A.R.V.I.S. is a local-first autonomous AI assistant designed to help with research, project work, code understanding, content generation, memory continuity, browser-safe workflows, and controlled automation.

This project blends:
- local AI/provider routing
- persistent memory
- project-aware file tools
- research and web search
- browser automation controls
- social/media campaign planning
- self-improvement workflows
- voice and HUD-driven interaction

A visual overview is available in the project website: [docs/index.html](docs/index.html)

## What JARVIS does

JARVIS is not just a chatbot. It is a project-aware assistant that can:

- answer questions with provider-aware routing
- search the web with Tavily
- inspect and modify files inside the project safely
- validate Python syntax and run tests
- generate social media briefs and branding concepts
- keep user preferences in memory over time
- analyze engagement and learn from performance results
- run controlled browser tasks only on approved domains
- operate within a structured safety model

## Important safety model

JARVIS is built with guardrails:

- file writes require explicit confirmation in safe mode
- browser automation is restricted to allowlisted domains
- social publishing requires approval before scheduling
- safety mode prevents risky actions unless you confirm them

This is important because the assistant is powerful, but it should stay controlled and human-reviewed where action is irreversible.

## Challenges and important realities

### 1. Safety must stay active
The biggest risk is giving the assistant too much autonomy without review. JARVIS should act safely and predictably.

### 2. Memory can become noisy
If memory is not structured well, the assistant may over-weight old data or recall the wrong preferences.

### 3. Tool use must stay scoped
JARVIS should only operate inside approved domains, approved folders, and explicitly confirmed actions.

### 4. Over-automation can reduce trust
The best system is not the most autonomous one; it is the one that is reliable, explainable, and reviewable.

## Recommendations for better use

- keep safety mode enabled until you are comfortable with an action
- treat memory like a long-term preference store, not a dump of everything
- give JARVIS clear goals and constraints
- review generated content before publishing
- use analytics memory to improve future creative decisions
